    18→const uid = (p: string) => `${p}${Math.random().toString(36).slice(2, 9)}${Date.now().toString(36).slice(-4)}`

    20→export function makeLayer(name: string, w: number, h: number, frameIds: string[], fill = true): Layer {
    21→  return {
    22→    id: uid('l'),
    23→    name,
    24→    visible: true,
    25→    locked: false,
    26→    opacity: 1,
    27→    cels: Object.fromEntries(frameIds.map((f) => [f, fill ? createCel(w, h) : null])),
    28→  }
    29→}
    30→
    31→export function makeFrame(durationMs = 120): Frame {
    32→  return { id: uid('f'), durationMs }
    33→}
    34→
    35→export function createDocument(w: number, h: number, name = 'Untitled'): PFDocument {
    36→  const frame = makeFrame(120)
    37→  const layer = makeLayer('Background', w, h, [frame.id])
    38→  return {
    39→    name, width: w, height: h,
    40→    frames: [frame],
    41→    layers: [layer],
    42→    palette: [...DEFAULT_PALETTE],
    43→    background: 'checker',
    44→    metadata: { createdAt: new Date().toISOString(), updatedAt: new Date().toISOString() },
    45→  }
    46→}
    47→
    48→function cloneDoc(doc: PFDocument): PFDocument {
    49→  return {
    50→    ...doc,
    51→    frames: doc.frames.map((f) => ({ ...f })),
    52→    layers: doc.layers.map((l) => ({
    53→      ...l,
    54→      cels: Object.fromEntries(Object.entries(l.cels).map(([k, v]) => [k, v ? cloneCel(v) : null])),
    55→    })),
    56→    palette: [...doc.palette],
    57→    metadata: { ...doc.metadata },
    58→  }
    59→}

   193→function snapDoc(label: string, mutate: (d: PFDocument) => void) {
   194→  const get = () => useEditorStore.getState()
   195→  const set = useEditorStore.setState
   196→  const { doc } = get()
   197→  const before = cloneDoc(doc)
   198→  const after = cloneDoc(doc)
   199→  mutate(after)
   200→  after.metadata.updatedAt = new Date().toISOString()
   201→  get().pushHistory({ kind: 'doc', label, before, after })
   202→  set({ doc: after, dirty: true, structureVersion: get().structureVersion + 1, pixelVersion: get().pixelVersion + 1 })
   203→}

   205→let setStore: (partial: Partial<EditorState> | ((s: EditorState) => Partial<EditorState>)) => void
   206→let getStore: () => EditorState
   207→
   208→export const useEditorStore = create<EditorState>((set, get) => {
   209→  setStore = set
   210→  getStore = get
   211→  const doc0 = createDocument(32, 32)
   212→  return {
   213→    doc: doc0,
   214→    activeFrameId: doc0.frames[0].id,
   215→    activeLayerId: doc0.layers[0].id,
   216→    tool: 'pencil',
   217→    toolOptions: { ...DEFAULT_TOOL_OPTIONS },
   218→    primary: '#000000',
   219→    secondary: '#FFFFFF',
   220→    recentColors: [],
   221→    selection: null,
   222→    floating: null,
   223→    clipboard: null,
   224→    view: { zoom: 8, panX: 0, panY: 0 },
   225→    viewport: { w: 800, h: 600 },
   226→    gridVisible: true,
   227→    onionSkin: false,
   228→    playing: false,
   229→    pixelVersion: 1,
   230→    structureVersion: 1,
   231→    dirty: false,
   232→    projectId: null,
   233→    strokeActive: false,
   234→    strokeButton: 0,
   235→    strokeAnchor: null,
   236→    strokeBefore: null,
   237→    strokeLabel: '',
   238→    lastPoint: null,
   239→    previewCel: null,
   240→    undoStack: [],
   241→    redoStack: [],
   242→    historyBytes: 0,
   243→
   244→    // ================= document =================
   245→    newDoc: (w, h, name) => {
   246→      const d = createDocument(w, h, name)
   247→      set({
   248→        doc: d, activeFrameId: d.frames[0].id, activeLayerId: d.layers[0].id,
   249→        selection: null, floating: null, previewCel: null, dirty: false, projectId: null,
   250→        undoStack: [], redoStack: [], historyBytes: 0, playing: false,
   251→        structureVersion: get().structureVersion + 1, pixelVersion: get().pixelVersion + 1,
   252→      })
   253→      get().fitToViewport()
   254→    },
   255→
   256→    loadDoc: (d, projectId) => {
   257→      set({
   258→        doc: d, activeFrameId: d.frames[0].id, activeLayerId: d.layers[d.layers.length - 1].id,
   259→        selection: null, floating: null, previewCel: null, dirty: false,
   260→        projectId: projectId ?? null, undoStack: [], redoStack: [], historyBytes: 0, playing: false,
   261→        structureVersion: get().structureVersion + 1, pixelVersion: get().pixelVersion + 1,
   262→      })
   263→      get().fitToViewport()
   264→    },
   265→
   266→    setDocName: (name) => { set({ doc: { ...get().doc, name }, dirty: true, structureVersion: get().structureVersion + 1 }) },
   267→    markSaved: (projectId) => set({ dirty: false, projectId: projectId ?? get().projectId }),
   268→    setBackground: (bg) => snapDoc('Canvas background', (d) => { d.background = bg }),
   269→
   270→    resizeCanvas: (w, h, anchor = 'center') => {
   271→      snapDoc('Resize canvas', (d) => {
   272→        const ow = d.width, oh = d.height
   273→        const dx = anchor.includes('w') ? 0 : anchor.includes('e') ? w - ow : Math.floor((w - ow) / 2)
   274→        const dy = anchor.includes('n') ? 0 : anchor.includes('s') ? h - oh : Math.floor((h - oh) / 2)
   275→        d.width = w; d.height = h
   276→        for (const layer of d.layers) {
   277→          for (const fid of Object.keys(layer.cels)) {
   278→            const cel = layer.cels[fid]
   279→            if (!cel) continue
   280→            const nc = createCel(w, h)
   281→            for (let y = 0; y < oh; y++) {
   282→              const ty = y + dy
   283→              if (ty < 0 || ty >= h) continue
   284→              for (let x = 0; x < ow; x++) {
   285→                const tx = x + dx
   286→                if (tx < 0 || tx >= w) continue
   287→                const si = (y * ow + x) * 4, di = (ty * w + tx) * 4
   288→                nc[di] = cel[si]; nc[di + 1] = cel[si + 1]; nc[di + 2] = cel[si + 2]; nc[di + 3] = cel[si + 3]
   289→              }
   290→            }
   291→            layer.cels[fid] = nc
   292→          }
   293→        }
   294→      })
   295→      set({ selection: null, floating: null, previewCel: null })
   296→      get().fitToViewport()
   297→    },
   298→
   299→    scaleArtwork: (factor) => {
   300→      snapDoc('Resize artwork', (d) => {
   301→        const nw = Math.max(1, Math.round(d.width * factor))
   302→        const nh = Math.max(1, Math.round(d.height * factor))
   303→        for (const layer of d.layers) {
   304→          for (const fid of Object.keys(layer.cels)) {
   305→            const cel = layer.cels[fid]
   306→            if (!cel) continue
   307→            const sc = nearestScale(cel, d.width, d.height, factor)
   308→            layer.cels[fid] = sc.data
   309→          }
   310→        }
   311→        d.width = nw; d.height = nh
   312→      })
   313→      set({ selection: null, floating: null, previewCel: null })
   314→      get().fitToViewport()
   315→    },
   316→
   317→    cropToContent: () => {
   318→      const { doc, activeFrameId } = get()
   319→      const comp = compositeFrame(doc, activeFrameId)
   320→      const bbox = computeBBox(comp, doc.width, doc.height)
   321→      if (!bbox) return
   322→      snapDoc('Trim', (d) => {
   323→        const { x, y, w, h } = bbox
   324→        for (const layer of d.layers) {
   325→          for (const fid of Object.keys(layer.cels)) {
   326→            const cel = layer.cels[fid]
   327→            if (!cel) continue
   328→            layer.cels[fid] = extractRegion(cel, d.width, d.height, { x, y, w, h })
   329→          }
   330→        }
   331→        d.width = w; d.height = h
   332→      })
   333→      set({ selection: null, floating: null })
   334→      get().fitToViewport()
   335→    },
   336→
   337→    cropToSelection: () => {
   338→      const sel = get().selection
   339→      if (!sel) return
   340→      snapDoc('Crop to selection', (d) => {
   341→        for (const layer of d.layers) {
   342→          for (const fid of Object.keys(layer.cels)) {
   343→            const cel = layer.cels[fid]
   344→            if (!cel) continue
   345→            layer.cels[fid] = extractRegion(cel, d.width, d.height, sel)
   346→          }
   347→        }
   348→        d.width = sel.w; d.height = sel.h
   349→      })
   350→      set({ selection: null, floating: null })
   351→      get().fitToViewport()
   352→    },
   353→
   354→    // ================= view =================
   355→    setZoom: (zoom, focus) => {
   356→      const z = Math.max(0.25, Math.min(32, zoom))
   357→      const { view } = get()
   358→      if (focus) {
   359→        const rx = (focus.x - view.panX) / view.zoom
   360→        const ry = (focus.y - view.panY) / view.zoom
   361→        set({ view: { zoom: z, panX: focus.x - rx * z, panY: focus.y - ry * z } })
   362→      } else {
   363→        set({ view: { ...view, zoom: z } })
   364→      }
   365→    },
   366→    zoomBy: (factor, focus) => get().setZoom(get().view.zoom * factor, focus),
   367→    panBy: (dx, dy) => set({ view: { ...get().view, panX: get().view.panX + dx, panY: get().view.panY + dy } }),
   368→    setPan: (x, y) => set({ view: { ...get().view, panX: x, panY: y } }),
   369→    setViewport: (w, h) => {
   370→      const changed = w !== get().viewport.w || h !== get().viewport.h
   371→      set({ viewport: { w, h } })
   372→      if (changed && get().view.zoom === get().view.zoom) {
   373→        // keep canvas visible if it is out of view
   374→        const { view, doc } = get()
   375→        const cw = doc.width * view.zoom, ch = doc.height * view.zoom
   376→        if (view.panX + cw < 0 || view.panY + ch < 0 || view.panX > w || view.panY > h) get().fitToViewport()
   377→      }
   378→    },
   379→    fitToViewport: () => {
   380→      const { viewport, doc } = get()
   381→      if (viewport.w < 10 || viewport.h < 10) return
   382→      const z = Math.max(0.25, Math.min(32, Math.min((viewport.w - 48) / doc.width, (viewport.h - 48) / doc.height)))
   383→      const snapped = z >= 1 ? Math.max(1, Math.floor(z * 2) / 2) : z
   384→      set({
   385→        view: {
   386→          zoom: snapped,
   387→          panX: Math.round((viewport.w - doc.width * snapped) / 2),
   388→          panY: Math.round((viewport.h - doc.height * snapped) / 2),
   389→        },
   390→      })
   391→    },
   392→    actualSize: () => {
   393→      const { viewport, doc } = get()
   394→      set({
   395→        view: {
   396→          zoom: 1,
   397→          panX: Math.round((viewport.w - doc.width) / 2),
   398→          panY: Math.round((viewport.h - doc.height) / 2),
   399→        },
   400→      })
   401→    },
   402→    centerCanvas: () => {
   403→      const { viewport, doc, view } = get()
   404→      set({ view: { ...view, panX: Math.round((viewport.w - doc.width * view.zoom) / 2), panY: Math.round((viewport.h - doc.height * view.zoom) / 2) } })
   405→    },
   406→    setGridVisible: (v) => set({ gridVisible: v }),
   407→    setOnionSkin: (v) => set({ onionSkin: v, pixelVersion: get().pixelVersion + 1 }),
   408→    setPlaying: (v) => set({ playing: v }),
   409→
   410→    // ================= tools =================
   411→    setTool: (t) => {
   412→      if (get().floating) get().commitFloating()
   413→      set({ tool: t, previewCel: null, strokeActive: false })
   414→    },
   415→    setToolOptions: (o) => set({ toolOptions: { ...get().toolOptions, ...o } }),
   416→    setPrimary: (hex, pushRecent = true) => {
   417→      const clean = hex.toUpperCase()
   418→      if (!pushRecent) { set({ primary: clean }); return }
   419→      const rec = [clean, ...get().recentColors.filter((c) => c !== clean)].slice(0, 24)
   420→      set({ primary: clean, recentColors: rec })
   421→    },
   422→    setSecondary: (hex) => set({ secondary: hex.toUpperCase() }),
   423→    swapColors: () => set({ primary: get().secondary, secondary: get().primary }),
   424→    resetColors: () => set({ primary: '#000000', secondary: '#FFFFFF' }),
   425→
   426→    // ================= strokes =================
   427→    canPaint: () => {
   428→      const { doc, activeLayerId } = get()
   429→      const layer = doc.layers.find((l) => l.id === activeLayerId)
   430→      return !!layer && layer.visible && !layer.locked
   431→    },
   432→
   433→    strokeBegin: (x, y, button, _shift) => {
   434→      const s = get()
   435→      if (x < 0 || y < 0 || x >= s.doc.width || y >= s.doc.height) return false
   436→      if (s.floating) s.commitFloating()
   437→      const layer = s.doc.layers.find((l) => l.id === s.activeLayerId)
   438→      if (!layer || !layer.visible || layer.locked) return false
   439→      const beforeCel = layer.cels[s.activeFrameId] ?? createCel(s.doc.width, s.doc.height)
   440→      const before = cloneCel(beforeCel)
   441→      const opts = { pattern: s.toolOptions.pattern, clip: s.selection }
   442→      const sym = s.toolOptions
   443→
   444→      set({
   445→        strokeActive: true,
   446→        strokeButton: button,
   447→        strokeAnchor: { x, y },
   448→        strokeBefore: before,
   449→        lastPoint: { x, y },
   450→        strokeLabel: s.tool,
   451→      })
   452→
   453→      const cel = layer.cels[s.activeFrameId] ?? createCel(s.doc.width, s.doc.height)
   454→      layer.cels[s.activeFrameId] = cel
   455→
   456→      switch (s.tool) {
   457→        case 'pencil': {
   458→          const colRgba = hexToRgba(button === 2 ? s.secondary : s.primary)
   459→          const w = s.doc.width, h = s.doc.height
   460→          const paintPt = (px: number, py: number) => {
   461→            const pts: [number, number][] = [[px, py]]
   462→            if (sym.symmetryH) pts.push([w - 1 - px, py])
   463→            if (sym.symmetryV) pts.push([px, h - 1 - py])
   464→            if (sym.symmetryH && sym.symmetryV) pts.push([w - 1 - px, h - 1 - py])
   465→            for (const [ax, ay] of pts) {
   466→              const i = (ay * w + ax) * 4
   467→              if (ax < 0 || ay < 0 || ax >= w || ay >= h) continue
   468→              if (opts.clip && (ax < opts.clip.x || ay < opts.clip.y || ax >= opts.clip.x + opts.clip.w || ay >= opts.clip.y + opts.clip.h)) continue
   469→              if (!patternOk(s.toolOptions.pattern, ax, ay)) continue
   470→              cel[i] = colRgba[0]; cel[i + 1] = colRgba[1]; cel[i + 2] = colRgba[2]; cel[i + 3] = colRgba[3]
   471→            }
   472→          }
   473→          paintPt(x, y)
   474→          set({ pixelVersion: get().pixelVersion + 1 })
   475→          return true
   476→        }
   477→        case 'eraser': {
   478→          const w = s.doc.width, h = s.doc.height
   479→          const paintPt = (px: number, py: number) => {
   480→            const pts: [number, number][] = [[px, py]]
   481→            if (sym.symmetryH) pts.push([w - 1 - px, py])
   482→            if (sym.symmetryV) pts.push([px, h - 1 - py])
   483→            if (sym.symmetryH && sym.symmetryV) pts.push([w - 1 - px, h - 1 - py])
   484→            for (const [ax, ay] of pts) {
   485→              const i = (ay * w + ax) * 4
   486→              if (ax < 0 || ay < 0 || ax >= w || ay >= h) continue
   487→              if (opts.clip && (ax < opts.clip.x || ay < opts.clip.y || ax >= opts.clip.x + opts.clip.w || ay >= opts.clip.y + opts.clip.h)) continue
   488→              cel[i] = 0; cel[i + 1] = 0; cel[i + 2] = 0; cel[i + 3] = 0
   489→            }
   490→          }
   491→          paintPt(x, y)
   492→          set({ pixelVersion: get().pixelVersion + 1 })
   493→          return true
   494→        }
   495→        case 'bucket': {
   496→          const col = hexToRgba(button === 2 ? s.secondary : s.primary)
   497→          floodFill(cel, s.doc.width, s.doc.height, x, y, col, s.toolOptions.tolerance, s.selection)
   498→          set({ pixelVersion: get().pixelVersion + 1 })
   499→          return true
   500→        }
   501→        case 'replace': {
   502→          const w = s.doc.width, h = s.doc.height
   503→          const i = (y * w + x) * 4
   504→          const from: RGBA = [cel[i], cel[i + 1], cel[i + 2], cel[i + 3]]
   505→          const to = hexToRgba(button === 2 ? s.secondary : s.primary)
   506→          replaceColor(cel, w, h, from, to, s.toolOptions.tolerance, s.selection)
   507→          set({ pixelVersion: get().pixelVersion + 1 })
   508→          return true
   509→        }
   510→        case 'line': case 'rect': case 'ellipse': {
   511→          const preview = createCel(s.doc.width, s.doc.height)
   512→          set({ previewCel: preview })
   513→          return true
   514→        }
   515→        case 'eyedropper': {
   516→          const comp = compositeFrame(s.doc, s.activeFrameId)
   517→          const i = (y * s.doc.width + x) * 4
   518→          const hex = '#' + [comp[i], comp[i + 1], comp[i + 2]].map((v) => v.toString(16).padStart(2, '0')).join('').toUpperCase()
   519→          if (button === 2) s.setSecondary(hex); else s.setPrimary(hex)
   520→          set({ strokeActive: false })
   521→          return true
   522→        }
   523→        case 'text': {
   524→          const t = s.toolOptions
   525→          if (!t.text.trim()) { set({ strokeActive: false }); return false }
   526→          const res = rasterizeText(s.doc.width, s.doc.height, t.text, t.textFont, t.textSize, x, y, hexToRgba(s.primary))
   527→          layer.cels[s.activeFrameId] = res.data
   528→          set({ pixelVersion: get().pixelVersion + 1, strokeActive: false })
   529→          get().pushHistory({
   530→            kind: 'pixels', label: 'Text', frameId: s.activeFrameId, layerId: s.activeLayerId,
   531→            before, after: cloneCel(res.data),
   532→          })
   533→          return false
   534→        }
   535→        case 'select': {
   536→          set({ previewCel: null })
   537→          return true
   538→        }
   539→        case 'move': {
   540→          const sel = s.selection
   541→          if (sel) {
   542→            const data = extractRegion(cel, s.doc.width, s.doc.height, sel)
   543→            clearRegion(cel, s.doc.width, s.doc.height, sel)
   544→            set({
   545→              floating: { data, w: sel.w, h: sel.h, x: sel.x, y: sel.y, originBefore: before, frameId: s.activeFrameId, layerId: s.activeLayerId },
   546→              pixelVersion: get().pixelVersion + 1,
   547→            })
   548→            return true
   549→          }
   550→          return false
   551→        }
   552→        default:
   553→          return false
   554→      }
   555→    },
   556→
   557→    strokeMove: (x, y, shift) => {
   558→      const s = get()
   559→      if (!s.strokeActive) return
   560→      const dx = x - (s.lastPoint?.x ?? x), dy = y - (s.lastPoint?.y ?? y)
   561→      set({ lastPoint: { x, y } })
   562→      const layer = s.doc.layers.find((l) => l.id === s.activeLayerId)
   563→      const w = s.doc.width, h = s.doc.height
   564→      const opts = { pattern: s.toolOptions.pattern, clip: s.selection }
   565→
   566→      if (s.tool === 'move' && s.floating) {
   567→        set({ floating: { ...s.floating, x: s.floating.x + dx, y: s.floating.y + dy }, pixelVersion: s.pixelVersion + 1 })
   568→        return
   569→      }
   570→      if (s.tool === 'select' && s.strokeAnchor) {
   571→        const a = s.strokeAnchor
   572→        set({
   573→          selection: {
   574→            x: Math.min(a.x, x), y: Math.min(a.y, y),
   575→            w: Math.abs(x - a.x) + 1, h: Math.abs(y - a.y) + 1,
   576→          },
   577→        })
   578→        return
   579→      }
   580→      if (!layer) return
   581→      const cel = layer.cels[s.activeFrameId]
   582→      if (!cel) return
   583→
   584→      if (s.tool === 'pencil' || s.tool === 'eraser') {
   585→        const prev = s.lastPoint
   586→        const colRgba = s.tool === 'eraser' ? ([0, 0, 0, 0] as RGBA) : hexToRgba(s.strokeButton === 2 ? s.secondary : s.primary)
   587→        if (prev) {
   588→          drawLine(cel, w, h, prev.x, prev.y, x, y, colRgba, opts)
   589→        }
   590→        set({ pixelVersion: s.pixelVersion + 1 })
   591→        return
   592→      }
   593→      if ((s.tool === 'line' || s.tool === 'rect' || s.tool === 'ellipse') && s.strokeAnchor && s.previewCel) {
   594→        const a = s.strokeAnchor
   595→        let ex = x, ey = y
   596→        if (shift && s.tool === 'line') {
   597→          const ddx = x - a.x, ddy = y - a.y
   598→          if (Math.abs(ddx) > Math.abs(ddy) * 2) ey = a.y
   599→          else if (Math.abs(ddy) > Math.abs(ddx) * 2) ex = a.x
   600→          else { const m = Math.min(Math.abs(ddx), Math.abs(ddy)); ex = a.x + Math.sign(ddx) * m; ey = a.y + Math.sign(ddy) * m }
   601→        }
   602→        const colRgba = hexToRgba(s.strokeButton === 2 ? s.secondary : s.primary)
   603→        const p = s.previewCel
   604→        p.fill(0)
   605→        if (s.tool === 'line') drawLine(p, w, h, a.x, a.y, ex, ey, colRgba, { clip: s.selection })
   606→        else if (s.tool === 'rect') drawRect(p, w, h, a.x, a.y, ex, ey, colRgba, s.toolOptions.shapeFilled, { clip: s.selection })
   607→        else drawEllipse(p, w, h, a.x, a.y, ex, ey, colRgba, s.toolOptions.shapeFilled, { clip: s.selection })
   608→        set({ pixelVersion: s.pixelVersion + 1 })
   609→      }
   610→    },
   611→
   612→    strokeEnd: (_shift) => {
   613→      const s = get()
   614→      if (!s.strokeActive) return
   615→      const layer = s.doc.layers.find((l) => l.id === s.activeLayerId)
   616→      const before = s.strokeBefore
   617→      set({ strokeActive: false, previewCel: null })
   618→      if (!layer || !before) return
   619→      const after = layer.cels[s.activeFrameId]
   620→      if (!after) return
   621→
   622→      if (s.tool === 'select') {
   623→        if (!s.selection || (s.selection.w <= 0 && s.selection.h <= 0)) set({ selection: null })
   624→        return
   625→      }
   626→      if (s.tool === 'eyedropper' || s.tool === 'pan') return
   627→      if (s.tool === 'move') return // floating continues to live until commit
   628→      if (s.tool === 'text') return // already pushed
   629→
   630→      let changed = false
   631→      for (let i = 0; i < before.length; i += 4) {
   632→        if (before[i] !== after[i] || before[i + 1] !== after[i + 1] || before[i + 2] !== after[i + 2] || before[i + 3] !== after[i + 3]) { changed = true; break }
   633→      }
   634→      if (!changed) { set({ strokeBefore: null }); return }
   635→      get().pushHistory({
   636→        kind: 'pixels',
   637→        label: s.tool === 'pencil' ? 'Pencil stroke' : s.tool === 'eraser' ? 'Erase' : s.tool,
   638→        frameId: s.activeFrameId,
   639→        layerId: s.activeLayerId,
   640→        before,
   641→        after: cloneCel(after),
   642→      })
   643→      set({ strokeBefore: null, dirty: true })
   644→    },
   645→
   646→    cancelStroke: () => {

   648→      if (s.strokeBefore) {
   649→        const layer = s.doc.layers.find((l) => l.id === s.activeLayerId)
   650→        if (layer) layer.cels[s.activeFrameId] = s.strokeBefore
   651→      }
   652→      set({ strokeActive: false, previewCel: null, strokeBefore: null, pixelVersion: s.pixelVersion + 1 })
   653→    },

   663→    moveFloating: (dx, dy) => {

   667→    },
   668→    commitFloating: () => {
   669→      const s = get()
   670→      const f = s.floating
   671→      if (!f) return
   672→      const layer = s.doc.layers.find((l) => l.id === f.layerId)
   673→      if (layer) {
   674→        const cel = layer.cels[f.frameId] ?? createCel(s.doc.width, s.doc.height)
   675→        layer.cels[f.frameId] = cel
   676→        for (let y = 0; y < f.h; y++) {
   677→          const ty = f.y + y
   678→          if (ty < 0 || ty >= s.doc.height) continue
   679→          for (let x = 0; x < f.w; x++) {

   684→              const di = (ty * s.doc.width + tx) * 4
   685→              cel[di] = f.data[si]; cel[di + 1] = f.data[si + 1]; cel[di + 2] = f.data[si + 2]; cel[di + 3] = f.data[si + 3]
   686→            }
   687→          }
   688→        }
   689→        const after = cloneCel(cel)
   690→        let changed = false
   691→        for (let i = 0; i < after.length; i += 4) {
   692→          if (f.originBefore[i] !== after[i] || f.originBefore[i + 1] !== after[i + 1] || f.originBefore[i + 2] !== after[i + 2] || f.originBefore[i + 3] !== after[i + 3]) { changed = true; break }
   693→        }
   694→        if (changed) {
   695→          get().pushHistory({ kind: 'pixels', label: 'Move', frameId: f.frameId, layerId: f.layerId, before: f.originBefore, after })
   696→        }
   697→      }
   698→      set({ floating: null, pixelVersion: get().pixelVersion + 1, dirty: true })
   699→    },
   700→    cutSelection: () => {
   701→      const s = get()
   702→      const sel = s.selection
   703→      if (!sel) return
   704→      if (s.floating) {
   705→        set({ clipboard: { data: cloneCel(s.floating.data), w: s.floating.w, h: s.floating.h }, floating: null, pixelVersion: s.pixelVersion + 1 })
   706→        return
   707→      }
   708→      const layer = s.doc.layers.find((l) => l.id === s.activeLayerId)
   709→      const cel = layer?.cels[s.activeFrameId]
   710→      if (!layer || !cel || layer.locked) return
   711→      set({ clipboard: { data: extractRegion(cel, s.doc.width, s.doc.height, sel), w: sel.w, h: sel.h } })
   712→      const before = cloneCel(cel)
   713→      clearRegion(cel, s.doc.width, s.doc.height, sel)
   714→      get().pushHistory({ kind: 'pixels', label: 'Cut', frameId: s.activeFrameId, layerId: s.activeLayerId, before, after: cloneCel(cel) })
   715→      set({ pixelVersion: s.pixelVersion + 1, dirty: true })
   716→    },
   717→    copySelection: () => {
   718→      const s = get()
   719→      const sel = s.selection
   720→      if (!sel) return
   721→      if (s.floating) {
   722→        set({ clipboard: { data: cloneCel(s.floating.data), w: s.floating.w, h: s.floating.h } })
   723→        return
   724→      }
   725→      const layer = s.doc.layers.find((l) => l.id === s.activeLayerId)
   726→      const cel = layer?.cels[s.activeFrameId]

   729→    },

   738→      const x = s.selection?.x ?? Math.floor((s.doc.width - cb.w) / 2)
   739→      const y = s.selection?.y ?? Math.floor((s.doc.height - cb.h) / 2)
   740→      set({
   741→        floating: { data: cloneCel(cb.data), w: cb.w, h: cb.h, x, y, originBefore: cloneCel(cel), frameId: s.activeFrameId, layerId: s.activeLayerId },
   742→        selection: { x, y, w: cb.w, h: cb.h },
   743→        pixelVersion: s.pixelVersion + 1,
   744→      })
   745→    },
   746→    deleteSelection: () => {
   747→      const s = get()
   748→      if (s.floating) {
   749→        set({ floating: null, pixelVersion: s.pixelVersion + 1 })
   750→        return
   751→      }
   752→      const sel = s.selection
   753→      if (!sel) return
   754→      const layer = s.doc.layers.find((l) => l.id === s.activeLayerId)
   755→      const cel = layer?.cels[s.activeFrameId]
   756→      if (!layer || !cel || layer.locked) return
   757→      const before = cloneCel(cel)
   758→      clearRegion(cel, s.doc.width, s.doc.height, sel)
   759→      get().pushHistory({ kind: 'pixels', label: 'Clear', frameId: s.activeFrameId, layerId: s.activeLayerId, before, after: cloneCel(cel) })
   760→      set({ pixelVersion: s.pixelVersion + 1, dirty: true })
   761→    },

   763→    // ================= layers =================
   764→    setActiveLayer: (id) => set({ activeLayerId: id }),
   765→    addLayer: () => {

   772→    },
   773→    deleteLayer: (id) => {
   774→      const s = get()
   775→      if (s.doc.layers.length <= 1) return
   776→      snapDoc('Delete layer', (d) => {
   777→        d.layers = d.layers.filter((l) => l.id !== id)
   778→      })
   779→      const st = get()
   780→      if (!st.doc.layers.find((l) => l.id === st.activeLayerId)) {
   781→        set({ activeLayerId: st.doc.layers[st.doc.layers.length - 1].id })
   782→      }
   783→    },
   784→    duplicateLayer: (id) => {
   785→      const s = get()
   786→      const layer = s.doc.layers.find((l) => l.id === id)
   787→      if (!layer) return
   788→      const copy: Layer = {
   789→        ...layer, id: uid('l'), name: `${layer.name} copy`,
   790→        cels: Object.fromEntries(Object.entries(layer.cels).map(([k, v]) => [k, v ? cloneCel(v) : null])),
   791→      }
   792→      const idx = s.doc.layers.findIndex((l) => l.id === id)
   793→      const layers = [...s.doc.layers]
   794→      layers.splice(idx + 1, 0, copy)
   795→      set({ doc: { ...s.doc, layers }, activeLayerId: copy.id, structureVersion: s.structureVersion + 1, dirty: true })
   796→    },
   797→    mergeDown: (id) => {
   798→      const s = get()
   799→      const idx = s.doc.layers.findIndex((l) => l.id === id)
   800→      if (idx <= 0) return
   801→      const top = s.doc.layers[idx], bottom = s.doc.layers[idx - 1]
   802→      snapDoc('Merge down', (d) => {
   803→        const t = d.layers[idx], b = d.layers[idx - 1]
   804→        for (const f of d.frames) {
   805→          const tc = t.cels[f.id]
   806→          const bc = b.cels[f.id] ?? createCel(d.width, d.height)
   807→          if (tc) {
   808→            for (let i = 0; i < bc.length; i += 4) {
   809→              const sa = (tc[i + 3] / 255) * t.opacity
   810→              if (sa <= 0) continue
   811→              const da = bc[i + 3] / 255
   812→              const ao = sa + da * (1 - sa)
   813→              if (ao <= 0) { bc[i] = bc[i + 1] = bc[i + 2] = bc[i + 3] = 0; continue }
   814→              bc[i] = (tc[i] * sa + bc[i] * da * (1 - sa)) / ao
   815→              bc[i + 1] = (tc[i + 1] * sa + bc[i + 1] * da * (1 - sa)) / ao
   816→              bc[i + 2] = (tc[i + 2] * sa + bc[i + 2] * da * (1 - sa)) / ao
   817→              bc[i + 3] = ao * 255
   818→            }
   819→          }
   820→          b.cels[f.id] = bc
   821→        }
   822→        b.visible = bottom.visible || top.visible
   823→        d.layers = d.layers.filter((l) => l.id !== id)
   824→      })
   825→      set({ activeLayerId: get().doc.layers[Math.max(0, idx - 1)].id })
   826→    },

   833→      const [l] = layers.splice(idx, 1)
   834→      layers.splice(to, 0, l)

   846→      })

   902→    },

   904→    // ================= transforms & ops =================
   905→    flip: (horizontal) => {
   906→      const s = get()
   907→      if (s.floating) {
   908→        const f = s.floating
   909→        const data = horizontal ? flipH(f.data, f.w, f.h) : flipV(f.data, f.w, f.h)
   910→        set({ floating: { ...f, data }, pixelVersion: s.pixelVersion + 1 })
   911→        return
   912→      }
   913→      const layer = s.doc.layers.find((l) => l.id === s.activeLayerId)
   914→      const cel = layer?.cels[s.activeFrameId]
   915→      if (!layer || !cel || layer.locked) return
   916→      const before = cloneCel(cel)
   917→      const sel = s.selection
   918→      if (sel) {
   919→        const region = extractRegion(cel, s.doc.width, s.doc.height, sel)
   920→        const flipped = horizontal ? flipH(region, sel.w, sel.h) : flipV(region, sel.w, sel.h)
   921→        clearRegion(cel, s.doc.width, s.doc.height, sel)
   922→        for (let y = 0; y < sel.h; y++) for (let x = 0; x < sel.w; x++) {
   923→          const si = (y * sel.w + x) * 4, di = ((sel.y + y) * s.doc.width + (sel.x + x)) * 4
   924→          cel[di] = flipped[si]; cel[di + 1] = flipped[si + 1]; cel[di + 2] = flipped[si + 2]; cel[di + 3] = flipped[si + 3]
   925→        }
   926→      } else {
   927→        layer.cels[s.activeFrameId] = horizontal ? flipH(cel, s.doc.width, s.doc.height) : flipV(cel, s.doc.width, s.doc.height)
   928→      }
   929→      get().pushHistory({
   930→        kind: 'pixels', label: horizontal ? 'Flip H' : 'Flip V', frameId: s.activeFrameId, layerId: s.activeLayerId,
   931→        before, after: cloneCel(layer.cels[s.activeFrameId] as Uint8ClampedArray),
   932→      })
   933→      set({ pixelVersion: s.pixelVersion + 1, dirty: true })
   934→    },
   935→
   936→    rotate: () => {
   937→      const s = get()
   938→      if (s.floating) {
   939→        const f = s.floating
   940→        const r = rotate90CW(f.data, f.w, f.h)
   941→        set({ floating: { ...f, data: r.data, w: r.w, h: r.h }, pixelVersion: s.pixelVersion + 1 })
   942→        return
   943→      }
   944→      if (s.selection) {
   945→        const layer = s.doc.layers.find((l) => l.id === s.activeLayerId)
   946→        const cel = layer?.cels[s.activeFrameId]
   947→        if (!layer || !cel || layer.locked) return
   948→        const sel = s.selection
   949→        const before = cloneCel(cel)
   950→        const region = extractRegion(cel, s.doc.width, s.doc.height, sel)
   951→        const r = rotate90CW(region, sel.w, sel.h)
   952→        clearRegion(cel, s.doc.width, s.doc.height, sel)
   953→        const newX = Math.max(0, Math.min(s.doc.width - r.w, sel.x))
   954→        const newY = Math.max(0, Math.min(s.doc.height - r.h, sel.y))
   955→        for (let y = 0; y < r.h; y++) for (let x = 0; x < r.w; x++) {
   956→          const si = (y * r.w + x) * 4, di = ((newY + y) * s.doc.width + (newX + x)) * 4
   957→          cel[di] = r.data[si]; cel[di + 1] = r.data[si + 1]; cel[di + 2] = r.data[si + 2]; cel[di + 3] = r.data[si + 3]
   958→        }
   959→        get().pushHistory({ kind: 'pixels', label: 'Rotate 90°', frameId: s.activeFrameId, layerId: s.activeLayerId, before, after: cloneCel(cel) })
   960→        set({ pixelVersion: s.pixelVersion + 1, dirty: true })
   961→        return
   962→      }
   963→      // rotate whole document
   964→      snapDoc('Rotate 90°', (d) => {
   965→        const ow = d.width, oh = d.height
   966→        for (const layer of d.layers) {
   967→          for (const fid of Object.keys(layer.cels)) {
   968→            const cel = layer.cels[fid]
   969→            if (!cel) continue
   970→            const rr = rotate90CW(cel, ow, oh)
   971→            layer.cels[fid] = rr.data
   972→          }
   973→        }
   974→        d.width = oh; d.height = ow
   975→      })
   976→      get().fitToViewport()
   977→    },
   978→
   979→    addOutline: () => {
   980→      const s = get()
   981→      const layer = s.doc.layers.find((l) => l.id === s.activeLayerId)
   982→      const cel = layer?.cels[s.activeFrameId]
   983→      if (!layer || !cel || layer.locked) return
   984→      const before = cloneCel(cel)
   985→      outlineCel(cel, s.doc.width, s.doc.height, hexToRgba(s.primary))
   986→      get().pushHistory({ kind: 'pixels', label: 'Outline', frameId: s.activeFrameId, layerId: s.activeLayerId, before, after: cloneCel(cel) })
   987→      set({ pixelVersion: s.pixelVersion + 1, dirty: true })
   988→    },
   989→
   990→    quantizeActiveLayer: (n) => {
   991→      const s = get()
   992→      const layer = s.doc.layers.find((l) => l.id === s.activeLayerId)
   993→      const cel = layer?.cels[s.activeFrameId]
   994→      if (!layer || !cel || layer.locked) return
   995→      const before = cloneCel(cel)
   996→      const q = quantizeLocal(cel, n)
   997→      layer.cels[s.activeFrameId] = q.data
   998→      get().pushHistory({ kind: 'pixels', label: `Reduce to ${n} colors`, frameId: s.activeFrameId, layerId: s.activeLayerId, before, after: cloneCel(q.data) })
   999→      if (q.palette.length > 0) {
  1000→        const merged = [...new Set([...q.palette, ...s.doc.palette])]
  1001→        set({ doc: { ...get().doc, palette: merged.slice(0, 256) }, structureVersion: get().structureVersion + 1 })
  1002→      }
  1003→      set({ pixelVersion: get().pixelVersion + 1, dirty: true })
  1004→    },
  1005→
  1006→    importImageIntoDoc: (data, w, h, mode) => {
  1007→      const s = get()
  1008→      if (mode === 'replace-layer') {
  1009→        const layer = s.doc.layers.find((l) => l.id === s.activeLayerId)
  1010→        if (!layer || layer.locked) return
  1011→        const cel = layer.cels[s.activeFrameId] ?? createCel(s.doc.width, s.doc.height)
  1012→        const before = cloneCel(cel)
  1013→        const nw = Math.min(s.doc.width, w), nh = Math.min(s.doc.height, h)
  1014→        for (let y = 0; y < nh; y++) for (let x = 0; x < nw; x++) {
  1015→          const si = (y * w + x) * 4, di = (y * s.doc.width + x) * 4
  1016→          cel[di] = data[si]; cel[di + 1] = data[si + 1]; cel[di + 2] = data[si + 2]; cel[di + 3] = data[si + 3]
  1017→        }
  1018→        get().pushHistory({ kind: 'pixels', label: 'Import image', frameId: s.activeFrameId, layerId: s.activeLayerId, before, after: cloneCel(cel) })
  1019→        set({ pixelVersion: s.pixelVersion + 1, dirty: true })
  1020→        return
  1021→      }
  1022→      // new layer sized to doc, image drawn centered/fit
  1023→      const layer = makeLayer(`Imported ${new Date().toLocaleTimeString()}`, s.doc.width, s.doc.height, [s.activeFrameId])
  1024→      const cel = createCel(s.doc.width, s.doc.height)
  1025→      const scale = Math.min(s.doc.width / w, s.doc.height / h, 1)
  1026→      const dw = Math.max(1, Math.round(w * scale)), dh = Math.max(1, Math.round(h * scale))
  1027→      const ox = Math.floor((s.doc.width - dw) / 2), oy = Math.floor((s.doc.height - dh) / 2)
  1028→      for (let y = 0; y < dh; y++) for (let x = 0; x < dw; x++) {
  1029→        const sx = Math.floor(x / scale), sy = Math.floor(y / scale)
  1030→        const si = (sy * w + sx) * 4, di = ((oy + y) * s.doc.width + (ox + x)) * 4
  1031→        cel[di] = data[si]; cel[di + 1] = data[si + 1]; cel[di + 2] = data[si + 2]; cel[di + 3] = data[si + 3]
  1032→      }
  1033→      layer.cels[s.activeFrameId] = cel
  1034→      const idx = s.doc.layers.findIndex((l) => l.id === s.activeLayerId)
  1035→      const layers = [...s.doc.layers]
  1036→      layers.splice(idx + 1, 0, layer)
  1037→      set({ doc: { ...s.doc, layers }, activeLayerId: layer.id, structureVersion: s.structureVersion + 1, pixelVersion: s.pixelVersion + 1, dirty: true })
  1038→    },
  1039→
  1040→    // ================= palette =================
  1041→    setDocPalette: (colors) => {
  1042→      set({ doc: { ...get().doc, palette: colors.slice(0, 256) }, structureVersion: get().structureVersion + 1, dirty: true })
  1043→    },
  1044→    addPaletteColor: (hex) => {
  1045→      const s = get()
  1046→      const c = hex.toUpperCase()
  1047→      if (s.doc.palette.includes(c) || s.doc.palette.length >= 256) return
  1048→      set({ doc: { ...s.doc, palette: [...s.doc.palette, c] }, structureVersion: s.structureVersion + 1, dirty: true })
  1049→    },
  1050→    removePaletteColor: (hex) => {
  1051→      const s = get()
  1052→      set({ doc: { ...s.doc, palette: s.doc.palette.filter((c) => c !== hex.toUpperCase()) }, structureVersion: s.structureVersion + 1, dirty: true })
  1053→    },
  1054→
  1055→    // ================= history =================
  1056→    undo: () => {
  1057→      const s = get()
  1058→      const entry = s.undoStack[s.undoStack.length - 1]
  1059→      if (!entry) return
  1060→      const undoStack = s.undoStack.slice(0, -1)
  1061→      if (entry.kind === 'pixels') {
  1062→        const layer = s.doc.layers.find((l) => l.id === entry.layerId)
  1063→        if (layer) layer.cels[entry.frameId] = cloneCel(entry.before)
  1064→        set({ undoStack, redoStack: [...s.redoStack, entry], pixelVersion: s.pixelVersion + 1, dirty: true })
  1065→      } else {
  1066→        const d = cloneDoc(entry.before)
  1067→        const activeLayerOk = d.layers.find((l) => l.id === s.activeLayerId)
  1068→        const activeFrameOk = d.frames.find((f) => f.id === s.activeFrameId)
  1069→        set({
  1070→          doc: d, undoStack, redoStack: [...s.redoStack, entry],
  1071→          activeLayerId: activeLayerOk ? s.activeLayerId : d.layers[d.layers.length - 1].id,
  1072→          activeFrameId: activeFrameOk ? s.activeFrameId : d.frames[0].id,
  1073→          structureVersion: s.structureVersion + 1, pixelVersion: s.pixelVersion + 1, dirty: true,
  1074→          selection: null, floating: null,
  1075→        })
  1076→      }
  1077→    },
  1078→
  1079→    redo: () => {
  1080→      const s = get()
  1081→      const entry = s.redoStack[s.redoStack.length - 1]
  1082→      if (!entry) return
  1083→      const redoStack = s.redoStack.slice(0, -1)
  1084→      if (entry.kind === 'pixels') {
  1085→        const layer = s.doc.layers.find((l) => l.id === entry.layerId)
  1086→        if (layer) layer.cels[entry.frameId] = cloneCel(entry.after)
  1087→        set({ redoStack, undoStack: [...s.undoStack, entry], pixelVersion: s.pixelVersion + 1, dirty: true })
  1088→      } else {
  1089→        const d = cloneDoc(entry.after)
  1090→        const activeLayerOk = d.layers.find((l) => l.id === s.activeLayerId)
  1091→        const activeFrameOk = d.frames.find((f) => f.id === s.activeFrameId)
  1092→        set({
  1093→          doc: d, redoStack, undoStack: [...s.undoStack, entry],
  1094→          activeLayerId: activeLayerOk ? s.activeLayerId : d.layers[d.layers.length - 1].id,
  1095→          activeFrameId: activeFrameOk ? s.activeFrameId : d.frames[0].id,
  1096→          structureVersion: s.structureVersion + 1, pixelVersion: s.pixelVersion + 1, dirty: true,
  1097→          selection: null, floating: null,
  1098→        })
  1099→      }
  1100→    },
  1101→
  1102→    pushHistory: (entry) => {
  1103→      const s = get()
  1104→      const bytes = entry.kind === 'pixels' ? entry.before.length * 2 : estimateDocBytes(entry.before)
  1105→      let undoStack = [...s.undoStack, entry]
  1106→      let historyBytes = s.historyBytes + bytes
  1107→      while (historyBytes > HISTORY_MAX_BYTES || undoStack.length > HISTORY_MAX_ENTRIES) {
  1108→        const removed = undoStack.shift()
  1109→        if (!removed) break
  1110→        historyBytes -= removed.kind === 'pixels' ? removed.before.length * 2 : estimateDocBytes(removed.before)
  1111→      }
  1112→      set({ undoStack, redoStack: [], historyBytes })
  1113→    },
  1114→  }
  1115→})
  1116→
  1117→// local import to avoid circular usage noise
  1118→import { quantize as quantizeLocal } from './engine'

  1120→function estimateDocBytes(doc: PFDocument): number {
  1121→  let bytes = 0
  1122→  for (const l of doc.layers) for (const cel of Object.values(l.cels)) if (cel) bytes += cel.length
  1123→  return bytes
  1124→}

  1126→function patternOk(pattern: string, x: number, y: number): boolean {
  1127→  switch (pattern) {
  1128→    case 'dither50': return (x + y) % 2 === 0
  1129→    case 'dither25': return x % 2 === 0 && y % 2 === 0
  1130→    case 'dither75': return !(x % 2 === 0 && y % 2 === 0)
  1131→    default: return true
  1132→  }
  1133→}
