   193→   193→function snapDoc(label: string, mutate: (d: PFDocument) => void) {
   194→   194→  const get = () => useEditorStore.getState()
   195→   195→  const set = useEditorStore.setState
   196→   196→  const { doc } = get()
   197→   197→  const before = cloneDoc(doc)
   198→   198→  const after = cloneDoc(doc)
   199→   199→  mutate(after)
   200→   200→  after.metadata.updatedAt = new Date().toISOString()
   201→   201→  get().pushHistory({ kind: 'doc', label, before, after })
   202→   202→  set({ doc: after, dirty: true, structureVersion: get().structureVersion + 1, pixelVersion: get().pixelVersion + 1 })
   203→   203→}

   205→   205→let setStore: (partial: Partial<EditorState> | ((s: EditorState) => Partial<EditorState>)) => void
   206→   206→let getStore: () => EditorState
   207→   207→
   208→   208→export const useEditorStore = create<EditorState>((set, get) => {
   209→   209→  setStore = set
   210→   210→  getStore = get
   211→   211→  const doc0 = createDocument(32, 32)
   212→   212→  return {
   213→   213→    doc: doc0,
   214→   214→    activeFrameId: doc0.frames[0].id,
   215→   215→    activeLayerId: doc0.layers[0].id,
   216→   216→    tool: 'pencil',
   217→   217→    toolOptions: { ...DEFAULT_TOOL_OPTIONS },
   218→   218→    primary: '#000000',
   219→   219→    secondary: '#FFFFFF',
   220→   220→    recentColors: [],
   221→   221→    selection: null,
   222→   222→    floating: null,
   223→   223→    clipboard: null,
   224→   224→    view: { zoom: 8, panX: 0, panY: 0 },
   225→   225→    viewport: { w: 800, h: 600 },
   226→   226→    gridVisible: true,
   227→   227→    onionSkin: false,
   228→   228→    playing: false,
   229→   229→    pixelVersion: 1,
   230→   230→    structureVersion: 1,
   231→   231→    dirty: false,
   232→   232→    projectId: null,
   233→   233→    strokeActive: false,
   234→   234→    strokeButton: 0,
   235→   235→    strokeAnchor: null,
   236→   236→    strokeBefore: null,
   237→   237→    strokeLabel: '',
   238→   238→    lastPoint: null,
   239→   239→    previewCel: null,
   240→   240→    undoStack: [],
   241→   241→    redoStack: [],
   242→   242→    historyBytes: 0,
   243→   243→
   244→   244→    // ================= document =================
   245→   245→    newDoc: (w, h, name) => {
   246→   246→      const d = createDocument(w, h, name)
   247→   247→      set({
   248→   248→        doc: d, activeFrameId: d.frames[0].id, activeLayerId: d.layers[0].id,
   249→   249→        selection: null, floating: null, previewCel: null, dirty: false, projectId: null,
   250→   250→        undoStack: [], redoStack: [], historyBytes: 0, playing: false,
   251→   251→        structureVersion: get().structureVersion + 1, pixelVersion: get().pixelVersion + 1,
   252→   252→      })
   253→   253→      get().fitToViewport()
   254→   254→    },
   255→   255→
   256→   256→    loadDoc: (d, projectId) => {
   257→   257→      set({
   258→   258→        doc: d, activeFrameId: d.frames[0].id, activeLayerId: d.layers[d.layers.length - 1].id,
   259→   259→        selection: null, floating: null, previewCel: null, dirty: false,
   260→   260→        projectId: projectId ?? null, undoStack: [], redoStack: [], historyBytes: 0, playing: false,
   261→   261→        structureVersion: get().structureVersion + 1, pixelVersion: get().pixelVersion + 1,
   262→   262→      })
   263→   263→      get().fitToViewport()
   264→   264→    },
   265→   265→
   266→   266→    setDocName: (name) => { set({ doc: { ...get().doc, name }, dirty: true, structureVersion: get().structureVersion + 1 }) },
   267→   267→    markSaved: (projectId) => set({ dirty: false, projectId: projectId ?? get().projectId }),
   268→   268→    setBackground: (bg) => snapDoc('Canvas background', (d) => { d.background = bg }),
   269→   269→
   270→   270→    resizeCanvas: (w, h, anchor = 'center') => {
   271→   271→      snapDoc('Resize canvas', (d) => {
   272→   272→        const ow = d.width, oh = d.height
   273→   273→        const dx = anchor.includes('w') ? 0 : anchor.includes('e') ? w - ow : Math.floor((w - ow) / 2)
   274→   274→        const dy = anchor.includes('n') ? 0 : anchor.includes('s') ? h - oh : Math.floor((h - oh) / 2)
   275→   275→        d.width = w; d.height = h
   276→   276→        for (const layer of d.layers) {
   277→   277→          for (const fid of Object.keys(layer.cels)) {
   278→   278→            const cel = layer.cels[fid]
   279→   279→            if (!cel) continue
   280→   280→            const nc = createCel(w, h)
   281→   281→            for (let y = 0; y < oh; y++) {
   282→   282→              const ty = y + dy
   283→   283→              if (ty < 0 || ty >= h) continue
   284→   284→              for (let x = 0; x < ow; x++) {
   285→   285→                const tx = x + dx
   286→   286→                if (tx < 0 || tx >= w) continue
   287→   287→                const si = (y * ow + x) * 4, di = (ty * w + tx) * 4
   288→   288→                nc[di] = cel[si]; nc[di + 1] = cel[si + 1]; nc[di + 2] = cel[si + 2]; nc[di + 3] = cel[si + 3]
   289→   289→              }
   290→   290→            }
   291→   291→            layer.cels[fid] = nc
   292→   292→          }
   293→   293→        }
   294→   294→      })
   295→   295→      set({ selection: null, floating: null, previewCel: null })
   296→   296→      get().fitToViewport()
   297→   297→    },
   298→   298→
   299→   299→    scaleArtwork: (factor) => {
   300→   300→      snapDoc('Resize artwork', (d) => {
   301→   301→        const nw = Math.max(1, Math.round(d.width * factor))
   302→   302→        const nh = Math.max(1, Math.round(d.height * factor))
   303→   303→        for (const layer of d.layers) {
   304→   304→          for (const fid of Object.keys(layer.cels)) {
   305→   305→            const cel = layer.cels[fid]
   306→   306→            if (!cel) continue
   307→   307→            const sc = nearestScale(cel, d.width, d.height, factor)
   308→   308→            layer.cels[fid] = sc.data
   309→   309→          }
   310→   310→        }
   311→   311→        d.width = nw; d.height = nh
   312→   312→      })
   313→   313→      set({ selection: null, floating: null, previewCel: null })
   314→   314→      get().fitToViewport()
   315→   315→    },
   316→   316→
   317→   317→    cropToContent: () => {
   318→   318→      const { doc, activeFrameId } = get()
   319→   319→      const comp = compositeFrame(doc, activeFrameId)
   320→   320→      const bbox = computeBBox(comp, doc.width, doc.height)
   321→   321→      if (!bbox) return
   322→   322→      snapDoc('Trim', (d) => {
   323→   323→        const { x, y, w, h } = bbox
   324→   324→        for (const layer of d.layers) {
   325→   325→          for (const fid of Object.keys(layer.cels)) {
   326→   326→            const cel = layer.cels[fid]
   327→   327→            if (!cel) continue
   328→   328→            layer.cels[fid] = extractRegion(cel, d.width, d.height, { x, y, w, h })
   329→   329→          }
   330→   330→        }
   331→   331→        d.width = w; d.height = h
   332→   332→      })
   333→   333→      set({ selection: null, floating: null })
   334→   334→      get().fitToViewport()
   335→   335→    },
   336→   336→
   337→   337→    cropToSelection: () => {
   338→   338→      const sel = get().selection
   339→   339→      if (!sel) return
   340→   340→      snapDoc('Crop to selection', (d) => {
   341→   341→        for (const layer of d.layers) {
   342→   342→          for (const fid of Object.keys(layer.cels)) {
   343→   343→            const cel = layer.cels[fid]
   344→   344→            if (!cel) continue
   345→   345→            layer.cels[fid] = extractRegion(cel, d.width, d.height, sel)
   346→   346→          }
   347→   347→        }
   348→   348→        d.width = sel.w; d.height = sel.h
   349→   349→      })
   350→   350→      set({ selection: null, floating: null })
   351→   351→      get().fitToViewport()
   352→   352→    },
   353→   353→
   354→   354→    // ================= view =================
   355→   355→    setZoom: (zoom, focus) => {
   356→   356→      const z = Math.max(0.25, Math.min(32, zoom))
   357→   357→      const { view } = get()
   358→   358→      if (focus) {
   359→   359→        const rx = (focus.x - view.panX) / view.zoom
   360→   360→        const ry = (focus.y - view.panY) / view.zoom
   361→   361→        set({ view: { zoom: z, panX: focus.x - rx * z, panY: focus.y - ry * z } })
   362→   362→      } else {
   363→   363→        set({ view: { ...view, zoom: z } })
   364→   364→      }
   365→   365→    },
   366→   366→    zoomBy: (factor, focus) => get().setZoom(get().view.zoom * factor, focus),
   367→   367→    panBy: (dx, dy) => set({ view: { ...get().view, panX: get().view.panX + dx, panY: get().view.panY + dy } }),
   368→   368→    setPan: (x, y) => set({ view: { ...get().view, panX: x, panY: y } }),
   369→   369→    setViewport: (w, h) => {
   370→   370→      const changed = w !== get().viewport.w || h !== get().viewport.h
   371→   371→      set({ viewport: { w, h } })
   372→   372→      if (changed && get().view.zoom === get().view.zoom) {
   373→   373→        // keep canvas visible if it is out of view
   374→   374→        const { view, doc } = get()
   375→   375→        const cw = doc.width * view.zoom, ch = doc.height * view.zoom
   376→   376→        if (view.panX + cw < 0 || view.panY + ch < 0 || view.panX > w || view.panY > h) get().fitToViewport()
   377→   377→      }
   378→   378→    },
   379→   379→    fitToViewport: () => {
   380→   380→      const { viewport, doc } = get()
   381→   381→      if (viewport.w < 10 || viewport.h < 10) return
   382→   382→      const z = Math.max(0.25, Math.min(32, Math.min((viewport.w - 48) / doc.width, (viewport.h - 48) / doc.height)))
   383→   383→      const snapped = z >= 1 ? Math.max(1, Math.floor(z * 2) / 2) : z
   384→   384→      set({
   385→   385→        view: {
   386→   386→          zoom: snapped,
   387→   387→          panX: Math.round((viewport.w - doc.width * snapped) / 2),
   388→   388→          panY: Math.round((viewport.h - doc.height * snapped) / 2),
   389→   389→        },
   390→   390→      })
   391→   391→    },
   392→   392→    actualSize: () => {
   393→   393→      const { viewport, doc } = get()
   394→   394→      set({
   395→   395→        view: {
   396→   396→          zoom: 1,
   397→   397→          panX: Math.round((viewport.w - doc.width) / 2),
   398→   398→          panY: Math.round((viewport.h - doc.height) / 2),
   399→   399→        },
   400→   400→      })
   401→   401→    },
   402→   402→    centerCanvas: () => {
   403→   403→      const { viewport, doc, view } = get()
   404→   404→      set({ view: { ...view, panX: Math.round((viewport.w - doc.width * view.zoom) / 2), panY: Math.round((viewport.h - doc.height * view.zoom) / 2) } })
   405→   405→    },
   406→   406→    setGridVisible: (v) => set({ gridVisible: v }),
   407→   407→    setOnionSkin: (v) => set({ onionSkin: v, pixelVersion: get().pixelVersion + 1 }),
   408→   408→    setPlaying: (v) => set({ playing: v }),
   409→   409→
   410→   410→    // ================= tools =================
   411→   411→    setTool: (t) => {
   412→   412→      if (get().floating) get().commitFloating()
   413→   413→      set({ tool: t, previewCel: null, strokeActive: false })
   414→   414→    },
   415→   415→    setToolOptions: (o) => set({ toolOptions: { ...get().toolOptions, ...o } }),
   416→   416→    setPrimary: (hex, pushRecent = true) => {
   417→   417→      const clean = hex.toUpperCase()
   418→   418→      if (!pushRecent) { set({ primary: clean }); return }
   419→   419→      const rec = [clean, ...get().recentColors.filter((c) => c !== clean)].slice(0, 24)
   420→   420→      set({ primary: clean, recentColors: rec })
   421→   421→    },
   422→   422→    setSecondary: (hex) => set({ secondary: hex.toUpperCase() }),
   423→   423→    swapColors: () => set({ primary: get().secondary, secondary: get().primary }),
   424→   424→    resetColors: () => set({ primary: '#000000', secondary: '#FFFFFF' }),
   425→   425→
   426→   426→    // ================= strokes =================
   427→   427→    canPaint: () => {
   428→   428→      const { doc, activeLayerId } = get()
   429→   429→      const layer = doc.layers.find((l) => l.id === activeLayerId)
   430→   430→      return !!layer && layer.visible && !layer.locked
   431→   431→    },
   432→   432→
   433→   433→    strokeBegin: (x, y, button, _shift) => {
   434→   434→      const s = get()
   435→   435→      if (x < 0 || y < 0 || x >= s.doc.width || y >= s.doc.height) return false
   436→   436→      if (s.floating) s.commitFloating()
   437→   437→      const layer = s.doc.layers.find((l) => l.id === s.activeLayerId)
   438→   438→      if (!layer || !layer.visible || layer.locked) return false
   439→   439→      const beforeCel = layer.cels[s.activeFrameId] ?? createCel(s.doc.width, s.doc.height)
   440→   440→      const before = cloneCel(beforeCel)
   441→   441→      const opts = { pattern: s.toolOptions.pattern, clip: s.selection }
   442→   442→      const sym = s.toolOptions
   443→   443→
   444→   444→      set({
   445→   445→        strokeActive: true,
   446→   446→        strokeButton: button,
   447→   447→        strokeAnchor: { x, y },

   449→   449→        lastPoint: { x, y },
   450→   450→        strokeLabel: s.tool,
   451→   451→      })
   452→   452→
   453→   453→      const cel = layer.cels[s.activeFrameId] ?? createCel(s.doc.width, s.doc.height)

   456→   456→      switch (s.tool) {

   463→   463→            if (sym.symmetryV) pts.push([px, h - 1 - py])
   464→   464→            if (sym.symmetryH && sym.symmetryV) pts.push([w - 1 - px, h - 1 - py])
   465→   465→            for (const [ax, ay] of pts) {
   466→   466→              const i = (ay * w + ax) * 4
   467→   467→              if (ax < 0 || ay < 0 || ax >= w || ay >= h) continue
   468→   468→              if (opts.clip && (ax < opts.clip.x || ay < opts.clip.y || ax >= opts.clip.x + opts.clip.w || ay >= opts.clip.y + opts.clip.h)) continue

   471→   471→            }
   472→   472→          }
   473→   473→          paintPt(x, y)
   474→   474→          set({ pixelVersion: get().pixelVersion + 1 })
   475→   475→          return true
   476→   476→        }
   477→   477→        case 'eraser': {
   478→   478→          const w = s.doc.width, h = s.doc.height
   479→   479→          const paintPt = (px: number, py: number) => {
   480→   480→            const pts: [number, number][] = [[px, py]]
   481→   481→            if (sym.symmetryH) pts.push([w - 1 - px, py])
   482→   482→            if (sym.symmetryV) pts.push([px, h - 1 - py])
   483→   483→            if (sym.symmetryH && sym.symmetryV) pts.push([w - 1 - px, h - 1 - py])
   484→   484→            for (const [ax, ay] of pts) {
   485→   485→              const i = (ay * w + ax) * 4
   486→   486→              if (ax < 0 || ay < 0 || ax >= w || ay >= h) continue
   487→   487→              if (opts.clip && (ax < opts.clip.x || ay < opts.clip.y || ax >= opts.clip.x + opts.clip.w || ay >= opts.clip.y + opts.clip.h)) continue
   488→   488→              cel[i] = 0; cel[i + 1] = 0; cel[i + 2] = 0; cel[i + 3] = 0
   489→   489→            }
   490→   490→          }
   491→   491→          paintPt(x, y)
   492→   492→          set({ pixelVersion: get().pixelVersion + 1 })
   493→   493→          return true
   494→   494→        }
   495→   495→        case 'bucket': {
   496→   496→          const col = hexToRgba(button === 2 ? s.secondary : s.primary)

   498→   498→          set({ pixelVersion: get().pixelVersion + 1 })
   499→   499→          return true
   500→   500→        }

   502→   502→          const w = s.doc.width, h = s.doc.height
   503→   503→          const i = (y * w + x) * 4
   504→   504→          const from: RGBA = [cel[i], cel[i + 1], cel[i + 2], cel[i + 3]]
   505→   505→          const to = hexToRgba(button === 2 ? s.secondary : s.primary)
   506→   506→          replaceColor(cel, w, h, from, to, s.toolOptions.tolerance, s.selection)
   507→   507→          set({ pixelVersion: get().pixelVersion + 1 })
   508→   508→          return true
   509→   509→        }
   510→   510→        case 'line': case 'rect': case 'ellipse': {
   511→   511→          const preview = createCel(s.doc.width, s.doc.height)
   512→   512→          set({ previewCel: preview })
   513→   513→          return true
   514→   514→        }
   515→   515→        case 'eyedropper': {

   517→   517→          const i = (y * s.doc.width + x) * 4

   520→   520→          set({ strokeActive: false })

   522→   522→        }

   529→   529→          get().pushHistory({
   530→   530→            kind: 'pixels', label: 'Text', frameId: s.activeFrameId, layerId: s.activeLayerId,
   531→   531→            before, after: cloneCel(res.data),
   532→   532→          })
   533→   533→          return false
   534→   534→        }
   535→   535→        case 'select': {
   536→   536→          set({ previewCel: null })
   537→   537→          return true
   538→   538→        }
   539→   539→        case 'move': {
   540→   540→          const sel = s.selection
   541→   541→          if (sel) {
   542→   542→            const data = extractRegion(cel, s.doc.width, s.doc.height, sel)
   543→   543→            clearRegion(cel, s.doc.width, s.doc.height, sel)
   544→   544→            set({
   545→   545→              floating: { data, w: sel.w, h: sel.h, x: sel.x, y: sel.y, originBefore: before, frameId: s.activeFrameId, layerId: s.activeLayerId },
   546→   546→              pixelVersion: get().pixelVersion + 1,
   547→   547→            })
   548→   548→            return true
   549→   549→          }
   550→   550→          return false
   551→   551→        }
   552→   552→        default:
   553→   553→          return false
   554→   554→      }
   555→   555→    },
   556→   556→
   557→   557→    strokeMove: (x, y, shift) => {

   579→   579→      }

   597→   597→          const ddx = x - a.x, ddy = y - a.y
   598→   598→          if (Math.abs(ddx) > Math.abs(ddy) * 2) ey = a.y
   599→   599→          else if (Math.abs(ddy) > Math.abs(ddx) * 2) ex = a.x
   600→   600→          else { const m = Math.min(Math.abs(ddx), Math.abs(ddy)); ex = a.x + Math.sign(ddx) * m; ey = a.y + Math.sign(ddy) * m }
   601→   601→        }
   602→   602→        const colRgba = hexToRgba(s.strokeButton === 2 ? s.secondary : s.primary)

   653→   653→    },

   693→   693→        }
   694→   694→        if (changed) {
   695→   695→          get().pushHistory({ kind: 'pixels', label: 'Move', frameId: f.frameId, layerId: f.layerId, before: f.originBefore, after })

   796→   796→    },
   797→   797→    mergeDown: (id) => {
   798→   798→      const s = get()
   799→   799→      const idx = s.doc.layers.findIndex((l) => l.id === id)
   800→   800→      if (idx <= 0) return
   801→   801→      const top = s.doc.layers[idx], bottom = s.doc.layers[idx - 1]
   802→   802→      snapDoc('Merge down', (d) => {
   803→   803→        const t = d.layers[idx], b = d.layers[idx - 1]
   804→   804→        for (const f of d.frames) {
   805→   805→          const tc = t.cels[f.id]
   806→   806→          const bc = b.cels[f.id] ?? createCel(d.width, d.height)

   808→   808→            for (let i = 0; i < bc.length; i += 4) {
   809→   809→              const sa = (tc[i + 3] / 255) * t.opacity
   810→   810→              if (sa <= 0) continue
   811→   811→              const da = bc[i + 3] / 255
   812→   812→              const ao = sa + da * (1 - sa)
   813→   813→              if (ao <= 0) { bc[i] = bc[i + 1] = bc[i + 2] = bc[i + 3] = 0; continue }
   814→   814→              bc[i] = (tc[i] * sa + bc[i] * da * (1 - sa)) / ao
   815→   815→              bc[i + 1] = (tc[i + 1] * sa + bc[i + 1] * da * (1 - sa)) / ao
   816→   816→              bc[i + 2] = (tc[i + 2] * sa + bc[i + 2] * da * (1 - sa)) / ao
   817→   817→              bc[i + 3] = ao * 255
   818→   818→            }
   819→   819→          }
   820→   820→          b.cels[f.id] = bc
   821→   821→        }
   822→   822→        b.visible = bottom.visible || top.visible
   823→   823→        d.layers = d.layers.filter((l) => l.id !== id)
   824→   824→      })

  1113→  1113→    },

  1120→  1120→function estimateDocBytes(doc: PFDocument): number {
  1121→  1121→  let bytes = 0
  1122→  1122→  for (const l of doc.layers) for (const cel of Object.values(l.cels)) if (cel) bytes += cel.length
  1123→  1123→  return bytes
  1124→  1124→}

  1126→  1126→function patternOk(pattern: string, x: number, y: number): boolean {
  1127→  1127→  switch (pattern) {
  1128→  1128→    case 'dither50': return (x + y) % 2 === 0
  1129→  1129→    case 'dither25': return x % 2 === 0 && y % 2 === 0
  1130→  1130→    case 'dither75': return !(x % 2 === 0 && y % 2 === 0)
  1131→  1131→    default: return true
  1132→  1132→  }
  1133→  1133→}
